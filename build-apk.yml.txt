name: 안드로이드 앱 만들기

on:
  workflow_dispatch:
  push:
    branches: [main, master]
    paths:
      - "android/**"
      - ".github/workflows/build-apk.yml"

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: 파일 가져오기
        uses: actions/checkout@v4

      - name: 자바 준비
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - name: 그레이들 준비
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: "8.9"

      - name: 앱 만들기
        working-directory: android
        run: gradle assembleRelease --no-daemon --stacktrace

      - name: 파일 이름 정리
        run: cp android/app/build/outputs/apk/release/app-release.apk radio.apk

      - name: 내려받기용 보관
        uses: actions/upload-artifact@v4
        with:
          name: radio-apk
          path: radio.apk

      - name: 공개 다운로드 주소 만들기
        uses: softprops/action-gh-release@v2
        with:
          tag_name: apk-latest
          name: 라방 앱 (최신)
          body: 라방 갤럭시 전용 앱 파일입니다. radio.apk 를 내려받아 설치하세요.
          files: radio.apk
          make_latest: true
